# RTX 50 (Blackwell, sm_120a) native Windows (MSVC) dev build, zbrad windows-dev. Experimental, not published.
# The same shape as requirements/rtx40.txt: setup.py reads this file as requirements/<GPU_TUNED_VARIANT>.txt (GPU_TUNED_VARIANT=rtx50win; the version label allows no hyphen) when
# VLLM_GB10_BUILD is set. torch and vllm_flash_attn are the local Windows builds, installed into the build venv
# first (torch 2.15.0+rtx50.cu134.win..., vllm_flash_attn 2.7.2.post1+rtx50.cu134.win...); flash-attention is not
# listed, as in rtx40.txt (VLLM_FLASH_ATTN_PREBUILT_PKG). flashinfer is left out until it is ported.

# Common dependencies
-r common.txt

# common.txt gates these on platform_machine == "x86_64", which is "AMD64" on Windows, so they would be dropped.
xgrammar == 0.2.8; sys_platform == "win32"
llguidance >= 1.7.0, < 1.8.0; sys_platform == "win32"

numba == 0.65.0 # Required for N-gram speculative decoding

# A plain version matches the local +rtx50.cu134.win... build (PEP 440 local versions).
torch == 2.15.0
# Triton for Windows (triton-lang/triton-windows), the version torch 2.15's checkout pins (3.9.0).
triton-windows == 3.9.0.post30; sys_platform == "win32"

# xgrammar 0.2.8's Windows wheel imports tvm::ffi::Error symbols that tvm_ffi.dll exports only from 0.1.13
# (0.1.11/0.1.12: WinError 127 on import). Every tilelang release caps apache-tvm-ffi < 0.1.13, so tilelang (the
# mhc kernels only) is left out on Windows; xgrammar is on the import path of every engine start.
apache-tvm-ffi==0.1.13.post3

# pyzmq with libzmq's ipc:// (AF_UNIX) transport on Windows: the zbrad/pyzmq +ipc build. PyPI's Windows wheels are
# built without it (pyzmq#1981), and vLLM then falls back to loopback TCP for the engine sockets (zmq.has("ipc")).
pyzmq @ https://github.com/zbrad/pyzmq/releases/download/v27.3.0%2Bipc.20261110.0/pyzmq-27.3.0%2Bipc.20261110.0-cp312-abi3-win_amd64.whl ; sys_platform == "win32"
# >=1.31.0 keeps execution plans alive for the CUDA graphs recorded from them
# (NVIDIA/cudnn-frontend#1195); older versions can replay an unloaded kernel.
nvidia-cudnn-frontend>=1.31.0
# Required for LLM_NVTX_SCOPES_FOR_PROFILING=1
nvtx==0.2.15
# Required for faster safetensors model loading
fastsafetensors >= 0.3.3

# QuACK and Cutlass DSL for FA4 (cute-DSL implementation)
nvidia-cutlass-dsl[cu13]==4.8.0
quack-kernels==0.6.5  # Required by MSA

# Humming kernels for quantization gemm
humming-kernels[cu13]==0.1.19
# Pin the pip CUDA components to the toolkit this build uses (cu134), as rtx40.txt does.
nvidia-cuda-runtime==13.4.92
nvidia-cuda-nvcc==13.4.92
nvidia-cuda-nvrtc==13.4.92
nvidia-nvvm==13.4.92
nvidia-cuda-crt==13.4.92
